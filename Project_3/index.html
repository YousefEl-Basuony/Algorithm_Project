📜 Algorithm Pseudocode Phase 1
1function heapSort(arr):
2  n = arr.length
3  // Phase 1: Build Heap
4  for i = ⌊n/2⌋ - 1 down to 0:
5    heapify(arr, n, i)
6  // Phase 2: In-place Extraction
7  for end = n - 1 down to 1:
8    swap(arr[0], arr[end])
9    // Sorted region grows; heap shrinks
10    heapify(arr, end, 0)
11function heapify(arr, heapSize, i):
12  largest = i, l = 2i + 1, r = 2i + 2
13  if l < heapSize && arr[l] > arr[largest]:
14  if r < heapSize && arr[r] > arr[largest]:
15  if largest != i:
16    swap(arr[i], arr[largest])
17    heapify(arr, heapSize, largest)
